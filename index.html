02 — Setup

Connect your Google Sheet

Pledges are written to your sheet through a small Google Apps Script "web app" you deploy once. This setting is shared with everyone using this tool, since it points at one shared sheet.

Web app URL

Not connected
You'll get this URL after deploying the script below.
How to set up the Google Sheet (one-time, ~5 minutes)
  1. Create or open the Google Sheet you want pledges saved into.
  2. Go to Extensions → Apps Script.
  3. Delete any starter code and paste in the script below.
  4. Click Deploy → New deployment. Choose type Web app.
  5. Set "Execute as" to Me, and "Who has access" to Anyone.
  6. Click Deploy, authorize it, then copy the web app URL it gives you.
  7. Paste that URL into the field above and save.
function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Pledges') || ss.insertSheet('Pledges'); var data = JSON.parse(e.postData.contents); if (sheet.getLastRow() === 0) { sheet.appendRow(['Scanned at', 'Donor name', 'Amount', 'Phone', 'Email', 'Address', 'Pledge date', 'Notes', 'Scanned by']); } sheet.appendRow([ new Date(), data.donorName || '', data.amount || '', data.phone || '', data.email || '', data.address || '', data.pledgeDate || '', data.notes || '', data.scannedBy || '' ]); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService.createTextOutput(JSON.stringify({ status: 'ok', message: 'Pledge Ledger endpoint is active' })) .setMimeType(ContentService.MimeType.JSON); }
